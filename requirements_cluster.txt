# --
# requirements for training-only runs on a GPU cluster (e.g. SLURM)
#
# This is requirements_pytorch.txt minus the packages a headless training
# node does not need and that may be hard to install there:
#   - PyQt6                    -> only an interactive matplotlib backend;
#                                 the cluster saves PNGs with the Agg backend
#                                 (set MPLBACKEND=Agg, see cluster/train.sbatch)
#   - ai-edge-quantizer-nightly-> only for int8 quantization (post-training,
#                                 no GPU); run that step locally instead
#                                 (BIODCASE_SKIP_QUANTIZATION=1 on the cluster)
#   - docker                   -> only for the ESP32 deployment toolchain,
#                                 which is skipped on the cluster
#
# litert-torch is kept so the float .tflite is still exported and evaluated.
# If litert-torch is also unavailable, training + .pth eval still run and the
# pipeline logs a warning and skips the tflite step.

flatbuffers
jinja2
librosa
litert-torch
matplotlib
numpy
numba
pyyaml
scipy
scikit-learn
soundfile
torch
torchinfo
tqdm
torchsummary
